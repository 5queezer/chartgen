import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Web app" icon="laptop">
    One chart per screen with drawings, alerts, watchlists and layouts, on a
    desktop or a phone. The Docker image serves it at `/` on the same origin as
    `/mcp`. See the [web guide](/chartgen/guides/web/).
  </Card>
  <Card title="70+ indicators" icon="seti:graphql">
    Overlays, momentum, volatility, volume and crypto metrics, plus your own
    [scripted indicators](/chartgen/guides/scripted-indicators/). See the
    [indicator reference](/chartgen/reference/indicators/).
  </Card>
  <Card title="Live data" icon="rocket">
    Yahoo Finance for stocks, Binance REST and WebSocket for crypto, with
    custom intervals such as `7m` or `2W`. Auto-detected from the ticker
    suffix.
  </Card>
  <Card title="TradingView parity" icon="star">
    See what the web chart supports, what is partial and what is missing in the
    [TradingView parity matrix](/chartgen/guides/tradingview-parity/).
  </Card>
  <Card title="MCP server" icon="puzzle">
    Connect Claude.ai to `/mcp` with [OAuth 2.1 PKCE](/chartgen/reference/oauth/),
    or run `--mcp` for Claude Desktop. 19 tools, plus optional scripting and
    server-alert tools. See the [MCP reference](/chartgen/reference/mcp/).
  </Card>
  <Card title="Trading engine" icon="setting">
    `--trade` runs a live WebSocket feed with price and indicator-signal
    [alerts](/chartgen/guides/trading/) and an order API that records orders
    locally.
  </Card>
</CardGrid>
